public sealed record DialogRow(
    string Name, int Width, int Height, int Attributes, string? Title, string FirstControl, string? DefaultControl,
    string? CancelControl);

public sealed record ControlRow(
    string Dialog, string Name, string Type, int X, int Y, int Width, int Height, int Attributes, string? Property,
    string? Text, string? Next);

public sealed record ConditionRow(string Dialog, string Control, string Action, string Condition);

public sealed record EventRow(string Dialog, string Control, string Event, string Argument, string Condition, int? Ordering);

public sealed record EventMappingRow(string Dialog, string Control, string Event, string Attribute);

public sealed record RadioButtonRow(string Property, int Order, string Value, int X, int Y, int Width, int Height, string? Text);

// A row of the ComboBox or ListBox table.
public sealed record ListItemRow(string Property, int Order, string Value, string? Text);

public sealed record CustomActionRow(string Action, int Type, string? Source, string? Target);

public sealed record SequenceRow(string Action, string? Condition, int Sequence);

public sealed record LaunchConditionRow(string Condition, string Description);

// Attributes 1 (msidbEmbeddedUI) marks the UI DLL, 0 a support file. No Data: only the attributes are read.
public sealed record EmbeddedUiRow(string Name, string FileName, int Attributes);

public sealed class PackageRows
{
    public List<(string Name, string Value)> Properties { get; } = [];
    public List<TextStyleRow> TextStyles { get; } = [];
    public List<DialogRow> Dialogs { get; } = [];
    public List<ControlRow> Controls { get; } = [];
    public List<ConditionRow> ControlConditions { get; } = [];
    public List<EventRow> ControlEvents { get; } = [];
    public List<EventMappingRow> EventMappings { get; } = [];
    public List<RadioButtonRow> RadioButtons { get; } = [];
    public List<ListItemRow> ComboBoxItems { get; } = [];
    public List<ListItemRow> ListBoxItems { get; } = [];
    public List<(string Name, byte[] Data)> Binaries { get; } = [];
    public List<CustomActionRow> CustomActions { get; } = [];
    public List<SequenceRow> UiSequence { get; } = [];
    public List<SequenceRow> ExecuteSequence { get; } = [];
    public List<LaunchConditionRow> LaunchConditions { get; } = [];
    public List<EmbeddedUiRow> EmbeddedUi { get; } = [];

    // Dialog, Control and TextStyle exist even when empty, as in any package with a UI.
    public bool WithUiTables { get; set; } = true;

    // Adds what Windows Installer checks before it opens a session on the package (MsiOpenPackage): the summary
    // stream and a ProductCode. Without them it refuses with 1620 and 1605.
    public bool InstallPackage { get; set; }
}

```csharp
namespace AdvancedInstallerMcp.Msi;

// Writes PackageRows into a new MSI with the standard table definitions (column types as in the Windows Installer
// docs). Used by mock_dialog (ADR 0008) and by the tests' generated packages.
public static class PackageWriter
{
    public static void Write(string path, PackageRows rows)
    {
        using var database = MsiDatabase.Create(path);

        database.Execute("CREATE TABLE `Property` (`Property` CHAR(72) NOT NULL, `Value` CHAR(0) NOT NULL LOCALIZABLE PRIMARY KEY `Property`)");
        foreach (var (name, value) in rows.Properties)
        {
            database.Execute("INSERT INTO `Property` (`Property`, `Value`) VALUES (?, ?)", name, value);
        }

        if (rows.InstallPackage)
        {
            database.Execute("INSERT INTO `Property` (`Property`, `Value`) VALUES (?, ?)", "ProductCode", NewGuid());
            database.SetPackageSummary(NewGuid());
        }

        WriteSequence(database, "InstallUISequence", rows.UiSequence);
        WriteSequence(database, "InstallExecuteSequence", rows.ExecuteSequence);

        if (rows.LaunchConditions.Count > 0)
        {
            database.Execute(
                "CREATE TABLE `LaunchCondition` (`Condition` CHAR(255) NOT NULL, `Description` CHAR(255) NOT NULL " +
                "LOCALIZABLE PRIMARY KEY `Condition`)");
            foreach (var row in rows.LaunchConditions)
            {
                database.Execute("INSERT INTO `LaunchCondition` (`Condition`, `Description`) VALUES (?, ?)", row.Condition, row.Description);
            }
        }

        if (rows.CustomActions.Count > 0)
        {
            database.Execute(
                "CREATE TABLE `CustomAction` (`Action` CHAR(72) NOT NULL, `Type` SHORT NOT NULL, `Source` CHAR(72), " +
                "`Target` CHAR(255) PRIMARY KEY `Action`)");
            foreach (var row in rows.CustomActions)
            {
                database.Execute("INSERT INTO `CustomAction` (`Action`, `Type`, `Source`, `Target`) VALUES (?, ?, ?, ?)",
                    row.Action, row.Type, row.Source, row.Target);
            }
        }

        if (rows.WithUiTables)
        {
            WriteUiTables(database, rows);
        }

        if (rows.EmbeddedUi.Count > 0)
        {
            database.Execute(
                "CREATE TABLE `MsiEmbeddedUI` (`MsiEmbeddedUI` CHAR(72) NOT NULL, `FileName` CHAR(255) NOT NULL, " +
                "`Attributes` SHORT NOT NULL, `MessageFilter` LONG, `Data` OBJECT PRIMARY KEY `MsiEmbeddedUI`)");
            foreach (var row in rows.EmbeddedUi)
            {
                database.Execute("INSERT INTO `MsiEmbeddedUI` (`MsiEmbeddedUI`, `FileName`, `Attributes`) VALUES (?, ?, ?)",
                    row.Name, row.FileName, row.Attributes);
            }
        }

        if (rows.Binaries.Count > 0)
        {
            database.Execute("CREATE TABLE `Binary` (`Name` CHAR(72) NOT NULL, `Data` OBJECT NOT NULL PRIMARY KEY `Name`)");
            foreach (var (name, data) in rows.Binaries)
            {
                // msi.dll takes a stream only from a file.
                var file = Path.GetTempFileName();
                try
                {
                    File.WriteAllBytes(file, data);
                    database.Execute("INSERT INTO `Binary` (`Name`, `Data`) VALUES (?, ?)", name, new FileInfo(file));
                }
                finally
                {
                    File.Delete(file);
                }
            }
        }

        database.Commit();
    }

    private static string NewGuid() => $"{{{Guid.NewGuid().ToString().ToUpperInvariant()}}}";

    private static void WriteSequence(MsiDatabase database, string table, List<SequenceRow> rows)
    {
        if (rows.Count == 0)
        {
            return;
        }

        database.Execute($"CREATE TABLE `{table}` (`Action` CHAR(72) NOT NULL, `Condition` CHAR(255), `Sequence` SHORT PRIMARY KEY `Action`)");
        foreach (var row in rows)
        {
            database.Execute($"INSERT INTO `{table}` (`Action`, `Condition`, `Sequence`) VALUES (?, ?, ?)", row.Action, row.Condition, row.Sequence);
        }
    }

    private static void WriteListItems(MsiDatabase database, string table, List<ListItemRow> rows)
    {
        if (rows.Count == 0)
        {
            return;
        }

        database.Execute(
            $"CREATE TABLE `{table}` (`Property` CHAR(72) NOT NULL, `Order` SHORT NOT NULL, `Value` CHAR(64) NOT NULL, " +
            "`Text` CHAR(64) LOCALIZABLE PRIMARY KEY `Property`, `Order`)");
        foreach (var row in rows)
        {
            database.Execute($"INSERT INTO `{table}` (`Property`, `Order`, `Value`, `Text`) VALUES (?, ?, ?, ?)",
                row.Property, row.Order, row.Value, row.Text);
        }
    }

    private static void WriteUiTables(MsiDatabase database, PackageRows rows)
    {
        // Copy CreateUiTables from TestPackage verbatim (TextStyle, Dialog, Control always; ControlCondition,
        // ControlEvent, RadioButton, EventMapping when they have rows), binding the record fields in column order:
        //   TextStyle: Name, FaceName, Size, Color, StyleBits
        //   Dialog: Name, 50, 50, Width, Height, Attributes, Title, FirstControl, DefaultControl, CancelControl
        //   Control: Dialog, Name, Type, X, Y, Width, Height, Attributes, Property, Text, Next, null (Help)
        //   ControlCondition: Dialog, Control, Action, Condition
        //   ControlEvent: Dialog, Control, Event, Argument, Condition, Ordering
        //   RadioButton: Property, Order, Value, X, Y, Width, Height, Text
        //   EventMapping: Dialog, Control, Event, Attribute
        // then:
        WriteListItems(database, "ComboBox", rows.ComboBoxItems);
        WriteListItems(database, "ListBox", rows.ListBoxItems);
    }
}
```

private ILookup<string, ListItem>? _comboItems;

// The entries of the ComboBox list tied to this property, in their Order.
public IReadOnlyList<ListItem> ComboItems(string property)
{
    _comboItems ??= _database.HasTable("ComboBox")
        ? _database.Query(
                "SELECT `Property`, `Order`, `Value`, `Text` FROM `ComboBox`",
                r => (Property: r.GetString(1), Order: r.GetInteger(2) ?? 0,
                    Item: new ListItem(r.GetString(3), r.GetString(4) is { Length: > 0 } text ? text : r.GetString(3))))
            .OrderBy(i => i.Order)
            .ToLookup(i => i.Property, i => i.Item, StringComparer.Ordinal)
        : Enumerable.Empty<ListItem>().ToLookup(_ => "", StringComparer.Ordinal);

    return _comboItems[property].ToList();
}

```csharp
using System.Text.RegularExpressions;
using AdvancedInstallerMcp.Msi;

namespace AdvancedInstallerMcp.Dialogs;

// Which property names the install itself uses (ADR 0009): named in a custom action scheduled in an execute
// sequence, in an execute-sequence condition or a LaunchCondition, or anywhere in a table that is not UI-only.
// A name counts as a whole word: DB_HOST is not used by [DB_HOST_NAME].
public sealed class InstallUsage
{
    private static readonly HashSet<string> UiOnlyTables = new(StringComparer.Ordinal)
    {
        "Dialog", "Control", "ControlEvent", "ControlCondition", "EventMapping", "RadioButton", "ComboBox", "ListBox",
        "ListView", "CheckBox", "TextStyle", "UIText", "Binary", "Icon", "Property", "InstallUISequence",
        "AdminUISequence", "_Validation", "_Tables", "_Columns", "_Streams", "_Storages",
    };

    private static readonly string[] ExecuteSequences = ["InstallExecuteSequence", "AdminExecuteSequence", "AdvtExecuteSequence"];

    // Windows Installer identifiers: letters, digits, underscores and periods.
    private static readonly Regex Word = new(@"[A-Za-z_][A-Za-z0-9_.]*", RegexOptions.CultureInvariant);

    private readonly HashSet<string> _used;
    private readonly HashSet<string> _directories;

    private InstallUsage(HashSet<string> used, HashSet<string> directories)
    {
        _used = used;
        _directories = directories;
    }

    public bool IsUsed(string property) => _used.Contains(property);

    public bool IsDirectory(string property) => _directories.Contains(property);

    public static InstallUsage Read(MsiDatabase database)
    {
        var used = new HashSet<string>(StringComparer.Ordinal);
        void AddWords(string cell)
        {
            foreach (Match match in Word.Matches(cell))
            {
                used.Add(match.Value);
            }
        }

        var tables = database.Query("SELECT `Name` FROM `_Tables`", r => r.GetString(1)).ToHashSet(StringComparer.Ordinal);
        var scheduled = new HashSet<string>(StringComparer.Ordinal);

        foreach (var sequence in ExecuteSequences.Where(tables.Contains))
        {
            foreach (var (action, condition) in database.Query($"SELECT `Action`, `Condition` FROM `{sequence}`",
                         r => (r.GetString(1), r.GetString(2))))
            {
                scheduled.Add(action);
                AddWords(condition);
            }
        }

        foreach (var table in tables.Where(t => !UiOnlyTables.Contains(t) && !ExecuteSequences.Contains(t)))
        {
            var columns = database.Query("SELECT `Name` FROM `_Columns` WHERE `Table` = ?", r => r.GetString(1), table);
            if (table == "CustomAction")
            {
                foreach (var cells in database.Query("SELECT `Action`, `Source`, `Target` FROM `CustomAction`",
                             r => (Action: r.GetString(1), Source: r.GetString(2), Target: r.GetString(3)))
                             .Where(a => scheduled.Contains(a.Action)))
                {
                    AddWords(cells.Source);
                    AddWords(cells.Target);
                }

                continue;
            }

            foreach (var column in columns)
            {
                // Integer and stream columns cannot be read as strings; their values hold no names.
                IReadOnlyList<string> cells;
                try
                {
                    cells = database.Query($"SELECT `{column}` FROM `{table}`", r => r.GetString(1));
                }
                catch (MsiException)
                {
                    continue;
                }

                foreach (var cell in cells)
                {
                    AddWords(cell);
                }
            }
        }

        var directories = tables.Contains("Directory")
            ? database.Query("SELECT `Directory` FROM `Directory`", r => r.GetString(1)).ToHashSet(StringComparer.Ordinal)
            : new HashSet<string>(StringComparer.Ordinal);

        return new InstallUsage(used, directories);
    }
}
```

Before relying on `catch (MsiException)`: check what `MsiRecord.GetString` does on an integer or stream column (read
`src/AdvancedInstallerMcp/Msi/MsiRecord.cs`). If it returns the number as text instead of throwing, the `try` is
harmless; if it throws something else, catch that type. A stream column (`Binary.Data`) is never read: `Binary` is
UI-only and no other standard table has a stream column, but an unknown table could — keep the catch.

```csharp
using AdvancedInstallerMcp.Dialogs;

namespace AdvancedInstallerMcp.Lint;

// ADR 0009: inputs on a page must reach the install and be settable without the page (msiexec /qn PROP=value).
public static class UnattendedRules
{
    private static readonly HashSet<string> InputTypes = new(StringComparer.Ordinal)
    {
        "Edit", "MaskedEdit", "PathEdit", "CheckBox", "RadioButtonGroup", "ComboBox", "ListBox",
    };

    // ComboBox: msidbControlAttributesComboList, a drop-down the user cannot type into.
    private const int ComboList = 0x00020000;

    private const int MaxRadioButtons = 5;
    private const int MaxComboListItemsForRadios = 3;

    public static bool IsInput(Control control) => InputTypes.Contains(control.Type) && control.Property is not null;

    public static IEnumerable<Finding> Check(DialogPackage package, Dialog dialog)
    {
        var secure = ListProperty(package, "SecureCustomProperties");
        var hidden = ListProperty(package, "MsiHiddenProperties");

        foreach (var control in dialog.Controls.Where(IsInput).OrderBy(c => c.Y).ThenBy(c => c.X))
        {
            var property = control.Property!;
            var isPrivate = property.Any(char.IsLower);
            var used = package.InstallUsage.IsUsed(property);

            if (used && isPrivate)
            {
                yield return new Finding("input-property-private", Severity.Error,
                    $"'{property}' has lower-case letters, so it is private: msiexec ignores it on the command line and " +
                    "Windows Installer does not pass it from the dialogs to the install, which uses it. Rename it in upper " +
                    $"case ('{property.ToUpperInvariant()}') everywhere it is used.",
                    dialog.Name, control.Name);
            }
            else if (used && !secure.Contains(property) && !package.InstallUsage.IsDirectory(property))
            {
                yield return new Finding("input-property-not-secure", Severity.Error,
                    $"The install uses '{property}', but it is not in SecureCustomProperties: when the install runs " +
                    "elevated, the value entered on this page is dropped and the install sees the default. Add it to " +
                    "SecureCustomProperties.",
                    dialog.Name, control.Name);
            }

            if (control.Type == "Edit" && control.Has(ControlAttributes.Password) && !hidden.Contains(property))
            {
                yield return new Finding("password-not-hidden", Severity.Warning,
                    $"'{property}' is a password field but not in MsiHiddenProperties, so a verbose log shows the value in " +
                    "clear text. Add it to MsiHiddenProperties.",
                    dialog.Name, control.Name);
            }

            var choices = control.Type switch
            {
                "RadioButtonGroup" => package.RadioButtons(property).Count,
                "ComboBox" => package.ComboItems(property).Count,
                _ => (int?)null,
            };

            if (choices is null)
            {
                continue;
            }

            if (!package.Properties.ContainsKey(property))
            {
                var radios = control.Type == "RadioButtonGroup";
                yield return new Finding("choice-without-default", radios ? Severity.Warning : Severity.Info,
                    (radios ? "No button is selected when the page opens" : "The list opens empty") +
                    $", and an unattended install without {property}=... passes an empty value that is none of the " +
                    $"choices. Add a Property row with the default value.",
                    dialog.Name, control.Name);
            }

            if (control.Type == "RadioButtonGroup" && choices > MaxRadioButtons)
            {
                yield return new Finding("control-choice", Severity.Info,
                    $"{choices} radio buttons: a drop-down list (ComboBox with the ComboList attribute) is easier to " +
                    "scan from 5 choices on.",
                    dialog.Name, control.Name);
            }
            else if (control.Type == "ComboBox" && control.Has(ComboList) && choices <= MaxComboListItemsForRadios)
            {
                yield return new Finding("control-choice", Severity.Info,
                    $"A drop-down list with {choices} entries hides choices that radio buttons would show at once.",
                    dialog.Name, control.Name);
            }
        }
    }

    private static HashSet<string> ListProperty(DialogPackage package, string name) =>
        (package.Properties.GetValueOrDefault(name) ?? "").Split(';', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries)
            .ToHashSet(StringComparer.Ordinal);
}
```

The `input-property-not-secure` message states that the value is dropped under elevation. ADR 0009's open point
(an elevated per-machine install) checks that together with the Directory exemption; if an administrator's UAC
install keeps the value, narrow the message to the case that drops it.

```csharp
using System.Text.RegularExpressions;

namespace AdvancedInstallerMcp.Mock;

// ADR 0008: a mock may only contain what Windows Installer can show. Every problem is listed, so one round trip fixes
// them all. Controls outside the dialog are not refused: lint_dialog reports them on the picture.
public static class MockSpecValidator
{
    public static readonly IReadOnlySet<string> ControlTypes = new SortedSet<string>(StringComparer.Ordinal)
    {
        "Billboard", "Bitmap", "CheckBox", "ComboBox", "DirectoryCombo", "DirectoryList", "Edit", "GroupBox",
        "Hyperlink", "Icon", "Line", "ListBox", "ListView", "MaskedEdit", "PathEdit", "ProgressBar", "PushButton",
        "RadioButtonGroup", "ScrollableText", "SelectionTree", "Text", "VolumeCostList", "VolumeSelectCombo",
    };

    public static readonly IReadOnlySet<string> InputTypes = new HashSet<string>(StringComparer.Ordinal)
    {
        "Edit", "MaskedEdit", "PathEdit", "CheckBox", "RadioButtonGroup", "ComboBox", "ListBox",
    };

    private static readonly HashSet<string> ItemTypes = new(StringComparer.Ordinal) { "RadioButtonGroup", "ComboBox", "ListBox" };

    private static readonly HashSet<string> ConditionActions = new(StringComparer.Ordinal) { "Show", "Hide", "Enable", "Disable", "Default" };

    private static readonly HashSet<string> Frames = new(StringComparer.Ordinal) { "wizard", "none" };

    private static readonly Regex Identifier = new("^[A-Za-z_][A-Za-z0-9_.]*$", RegexOptions.CultureInvariant);

    public static IReadOnlyList<string> Problems(MockSpec spec)
    {
        var problems = new List<string>();
        var frameNames = WizardFrame.Load().Controls.Select(c => c.Name).ToHashSet(StringComparer.Ordinal);

        void CheckName(string what, string name, int maxLength)
        {
            if (!Identifier.IsMatch(name) || name.Length > maxLength)
            {
                problems.Add($"{what} '{name}' is not a Windows Installer identifier (letters, digits, '_' and '.', " +
                    $"starting with a letter or '_', at most {maxLength} characters).");
            }
        }

        foreach (var style in spec.TextStyles ?? [])
        {
            CheckName("Text style", style.Name, 72);
        }

        foreach (var name in (spec.Properties ?? []).Keys)
        {
            CheckName("Property", name, 72);
        }

        foreach (var duplicate in spec.Dialogs.GroupBy(d => d.Name).Where(g => g.Count() > 1))
        {
            problems.Add($"Dialog '{duplicate.Key}' appears {duplicate.Count()} times.");
        }

        foreach (var dialog in spec.Dialogs)
        {
            CheckName("Dialog", dialog.Name, 72);
            if (!Frames.Contains(dialog.Frame))
            {
                problems.Add($"Dialog '{dialog.Name}': frame '{dialog.Frame}' is not 'wizard' or 'none'.");
            }

            var names = new HashSet<string>(StringComparer.Ordinal);
            foreach (var control in dialog.Controls)
            {
                var where = $"{dialog.Name}.{control.Name}";
                CheckName("Control", control.Name, 50);

                if (!names.Add(control.Name))
                {
                    problems.Add($"{where}: the control name appears twice on the page.");
                }

                if (dialog.Frame == "wizard" && frameNames.Contains(control.Name))
                {
                    problems.Add($"{where}: '{control.Name}' is a control of the wizard frame; pick another name.");
                }

                if (!ControlTypes.Contains(control.Type))
                {
                    problems.Add($"{where}: '{control.Type}' is not a Windows Installer control type. Use one of: " +
                        $"{string.Join(", ", ControlTypes)}.");
                    continue;
                }

                if (control.Property is { } property)
                {
                    CheckName($"{where}: property", property, 72);
                    if (InputTypes.Contains(control.Type) && property.Any(char.IsLower))
                    {
                        problems.Add($"{where}: '{property}' has lower-case letters, so it is private and cannot be " +
                            $"set on the msiexec command line. Use upper case ('{property.ToUpperInvariant()}').");
                    }
                }
                else if (InputTypes.Contains(control.Type))
                {
                    problems.Add($"{where}: a {control.Type} needs the property it sets.");
                }

                if (control.Items is { Count: > 0 } && !ItemTypes.Contains(control.Type))
                {
                    problems.Add($"{where}: only RadioButtonGroup, ComboBox and ListBox take items.");
                }
                else if (control.Type == "RadioButtonGroup" && control.Items is not { Count: > 0 })
                {
                    problems.Add($"{where}: a RadioButtonGroup needs its buttons in 'items'.");
                }
            }

            foreach (var condition in dialog.Conditions ?? [])
            {
                if (!names.Contains(condition.Control) && !(dialog.Frame == "wizard" && frameNames.Contains(condition.Control)))
                {
                    problems.Add($"{dialog.Name}: condition on '{condition.Control}', which is not on the page.");
                }

                if (!ConditionActions.Contains(condition.Action))
                {
                    problems.Add($"{dialog.Name}.{condition.Control}: condition action '{condition.Action}' is not one of " +
                        $"{string.Join(", ", ConditionActions)}.");
                }
            }
        }

        return problems;
    }
}
```

**Interfaces:**

Rules the builder applies:
- `properties` from the frame (`DefaultUIFont`), then `spec.Properties`, then `MockBannerTitle` / `MockBannerText`
  are **not** properties: the banner texts are written into the frame's two Text controls by replacing
  `[MockBannerTitle]` / `[MockBannerText]` in their `Text` with the page's `BannerTitle` / `BannerText` (empty when
  null). Per page, so each dialog gets its own copy of the frame controls.
- Text styles: frame's, then `spec.TextStyles` (spec wins on a name clash).
- Banner bitmap: one `Binary` row named after the frame Bitmap's `Text`, holding a plain white 24-bit BMP of the
  bitmap's size at 96 DPI (generated with `System.Drawing.Bitmap` → `ImageFormat.Bmp`).
- Attributes default 3. `Title` default the frame's. Width/Height default the frame's (370 × 270 for `none` too).
- Radio items: `RadioButton` rows; an item without position gets `X = 0, Y = 15 * index, Width = control.Width, Height = 15`.
  Combo items → `ComboBoxItems`, list items → `ListBoxItems`; `Order` = index + 1.
- Tab order when `Next` is omitted: body controls whose type is focusable (same set as `LayoutRules`' focusable
  types — make `LayoutRules.FocusableTypes` `internal static` and reuse it), in spec order, then the frame's own
  chain starting at its `FirstControl`; the last frame control in that chain whose `Next` pointed back to the frame's
  first now points to the first body control. With frame `none`, the body controls form a cycle on their own.
  `FirstControl` = first focusable body control, else the frame's.
- `DefaultControl` / `CancelControl`: the frame's (`none`: null).

using System.Drawing;
using System.Drawing.Imaging;
using AdvancedInstallerMcp.Dialogs;
using AdvancedInstallerMcp.Lint;
using AdvancedInstallerMcp.Msi;

namespace AdvancedInstallerMcp.Mock;

// Turns a validated MockSpec into the rows of a throw-away package (ADR 0008). Wizard pages get their own copy of
// Advanced Installer's frame, with the page's banner texts and the body's controls tabbed in before the buttons.
public static class MockPackageBuilder
{
    private const int DefaultAttributes = ControlAttributes.Visible | ControlAttributes.Enabled;
    private const int RadioButtonHeight = 15;

    public static PackageRows Build(MockSpec spec)
    {
        var frame = WizardFrame.Load();
        var rows = new PackageRows();

        var properties = new Dictionary<string, string>(frame.Properties, StringComparer.Ordinal);
        foreach (var (name, value) in spec.Properties ?? [])
        {
            properties[name] = value;
        }

        rows.Properties.AddRange(properties.Select(p => (p.Key, p.Value)));

        var styles = frame.TextStyles.ToDictionary(s => s.Name, StringComparer.Ordinal);
        foreach (var style in spec.TextStyles ?? [])
        {
            styles[style.Name] = new TextStyleRow(style.Name, style.FaceName, style.Size, style.Color, style.StyleBits);
        }

        rows.TextStyles.AddRange(styles.Values);

        if (spec.Dialogs.Any(d => d.Frame == "wizard"))
        {
            var bitmap = frame.Controls.Single(c => c.Name == frame.BannerBitmap);
            rows.Binaries.Add((bitmap.Text!, PlainBitmap(bitmap.Width, bitmap.Height)));
        }

        foreach (var dialog in spec.Dialogs)
        {
            AddDialog(rows, frame, dialog);
        }

        return rows;
    }

    private static void AddDialog(PackageRows rows, WizardFrame frame, MockDialogSpec dialog)
    {
        var wizard = dialog.Frame == "wizard";
        var body = dialog.Controls
            .Select(c => new ControlRow(dialog.Name, c.Name, c.Type, c.X, c.Y, c.Width, c.Height,
                c.Attributes ?? DefaultAttributes, c.Property, c.Text, c.Next))
            .ToList();
        var frameRows = wizard
            ? frame.Controls.Select(c => new ControlRow(dialog.Name, c.Name, c.Type, c.X, c.Y, c.Width, c.Height,
                c.Attributes, c.Property, BannerText(frame, c, dialog), c.Next)).ToList()
            : [];

        var focusable = body.Where(c => c.Next is null && LayoutRules.FocusableTypes.Contains(c.Type)).ToList();
        var first = focusable.FirstOrDefault()?.Name ?? (wizard ? frame.FirstControl : body.FirstOrDefault()?.Name ?? "");
        Chain(focusable, frameRows, wizard ? frame.FirstControl : null, body);

        rows.Dialogs.Add(new DialogRow(dialog.Name, dialog.Width ?? frame.Width, dialog.Height ?? frame.Height,
            DefaultAttributes, dialog.Title ?? frame.Title, first,
            wizard ? frame.DefaultControl : null, wizard ? frame.CancelControl : null));
        rows.Controls.AddRange(body);
        rows.Controls.AddRange(frameRows);

        foreach (var condition in dialog.Conditions ?? [])
        {
            rows.ControlConditions.Add(new ConditionRow(dialog.Name, condition.Control, condition.Action, condition.Condition));
        }

        foreach (var control in dialog.Controls.Where(c => c.Items is { Count: > 0 }))
        {
            AddItems(rows, control);
        }
    }

    // Body controls tab in spec order; the last one hands over to the frame's chain, and the frame control that
    // closed the frame's cycle now points at the first body control.
    private static void Chain(List<ControlRow> focusable, List<ControlRow> frameRows, string? frameFirst, List<ControlRow> body)
    {
        if (focusable.Count == 0)
        {
            return;
        }

        for (var i = 0; i < focusable.Count; i++)
        {
            var next = i + 1 < focusable.Count ? focusable[i + 1].Name : frameFirst ?? focusable[0].Name;
            body[body.IndexOf(focusable[i])] = focusable[i] with { Next = next };
        }

        if (frameFirst is null)
        {
            return;
        }

        var closing = frameRows.FindIndex(c => c.Next == frameFirst);
        if (closing >= 0)
        {
            frameRows[closing] = frameRows[closing] with { Next = focusable[0].Name };
        }
    }

    private static string? BannerText(WizardFrame frame, FrameControl control, MockDialogSpec dialog) =>
        control.Name == frame.BannerTitle ? control.Text?.Replace("[MockBannerTitle]", dialog.BannerTitle ?? "")
        : control.Name == frame.BannerText ? control.Text?.Replace("[MockBannerText]", dialog.BannerText ?? "")
        : control.Text;

    private static void AddItems(PackageRows rows, MockControl control)
    {
        var items = control.Items!;
        for (var i = 0; i < items.Count; i++)
        {
            var item = items[i];
            switch (control.Type)
            {
                case "RadioButtonGroup":
                    rows.RadioButtons.Add(new RadioButtonRow(control.Property!, i + 1, item.Value,
                        item.X ?? 0, item.Y ?? RadioButtonHeight * i, item.Width ?? control.Width, item.Height ?? RadioButtonHeight,
                        item.Text));
                    break;
                case "ComboBox":
                    rows.ComboBoxItems.Add(new ListItemRow(control.Property!, i + 1, item.Value, item.Text));
                    break;
                case "ListBox":
                    rows.ListBoxItems.Add(new ListItemRow(control.Property!, i + 1, item.Value, item.Text));
                    break;
            }
        }
    }

    // Stands in for the banner picture, which is Advanced Installer's and not ours to ship.
    private static byte[] PlainBitmap(int widthUnits, int heightUnits)
    {
        using var bitmap = new Bitmap(
            Math.Max(1, InstallerUnits.ToPixels(widthUnits, 96)), Math.Max(1, InstallerUnits.ToPixels(heightUnits, 96)),
            PixelFormat.Format24bppRgb);
        using (var graphics = Graphics.FromImage(bitmap))
        {
            graphics.Clear(Color.White);
        }

        using var stream = new MemoryStream();
        bitmap.Save(stream, ImageFormat.Bmp);
        return stream.ToArray();
    }
}

using System.Net;
using System.Text;
using AdvancedInstallerMcp.Dialogs;
using AdvancedInstallerMcp.Lint;

namespace AdvancedInstallerMcp.Mock;

// The page the user approves before the project changes (ADR 0008): every proposed page, its inputs and how each is
// set without the UI. Generated from the spec, so its format never varies.
public static class ApprovalPage
{
    private const string Style =
        "body{font:14px/1.45 'Segoe UI',system-ui,sans-serif;margin:0 auto;max-width:880px;padding:24px 16px;color:#1f1f1f;background:#fafafa}" +
        "section{background:#fff;border:1px solid #ddd;border-radius:6px;padding:16px;margin:0 0 24px}" +
        "img{display:block;max-width:100%;border:1px solid #ccc;margin:8px 0}" +
        "table{border-collapse:collapse;width:100%;margin:8px 0}th,td{border:1px solid #ddd;padding:4px 8px;text-align:left;vertical-align:top}" +
        "code,pre{font:13px Consolas,monospace}pre{white-space:pre-wrap;background:#f3f3f3;padding:8px;border-radius:4px}" +
        ".Error{color:#b00020}.Warning{color:#8a5a00}.Info{color:#555}";

    public static string Html(MockSpec spec, IReadOnlyList<Finding> findings)
    {
        var html = new StringBuilder();
        html.Append("<!doctype html><html lang=\"en\"><head><meta charset=\"utf-8\"><meta name=\"viewport\" content=\"width=device-width,initial-scale=1\">")
            .Append("<title>Setup pages for review</title><style>").Append(Style).Append("</style></head><body>")
            .Append("<h1>Setup pages for review</h1>");

        foreach (var dialog in spec.Dialogs)
        {
            html.Append("<section><h2>").Append(Encode(dialog.Name)).Append("</h2>")
                .Append("<img src=\"").Append(Encode(dialog.Name)).Append(".png\" alt=\"").Append(Encode(dialog.Name)).Append("\">");
            if (dialog.Description is { } description)
            {
                html.Append("<p>").Append(Encode(description)).Append("</p>");
            }

            var inputs = dialog.Controls.Where(IsInput).ToList();
            if (inputs.Count > 0)
            {
                html.Append("<table><tr><th>Control</th><th>Property</th><th>Default</th><th>Why this control</th><th>Command line</th></tr>");
                foreach (var control in inputs)
                {
                    var property = control.Property!;
                    html.Append("<tr><td>").Append(Encode(control.Type)).Append("</td><td><code>").Append(Encode(property))
                        .Append("</code></td><td>").Append(Encode(Default(spec, property) ?? "(empty)"))
                        .Append("</td><td>").Append(Encode(control.Rationale ?? ""))
                        .Append("</td><td>").Append(string.Join("<br>", Forms(control).Select(Encode))).Append("</td></tr>");
                }

                html.Append("</table>");
            }

            var own = findings.Where(f => f.Dialog == dialog.Name).ToList();
            if (own.Count > 0)
            {
                html.Append("<ul>");
                foreach (var finding in own)
                {
                    html.Append("<li class=\"").Append(finding.Severity).Append("\"><code>").Append(Encode(finding.Rule)).Append("</code> ")
                        .Append(Encode(finding.Control is null ? "" : finding.Control + ": ")).Append(Encode(finding.Message)).Append("</li>");
                }

                html.Append("</ul>");
            }

            html.Append("</section>");
        }

        html.Append("<section><h2>Unattended install</h2><pre>").Append(Encode(CommandLine(spec))).Append("</pre></section>")
            .Append("</body></html>");
        return html.ToString();
    }

    public static string CommandLine(MockSpec spec)
    {
        var parts = new List<string> { "msiexec /i Setup.msi /qn /l*v install.log" };
        var seen = new HashSet<string>(StringComparer.Ordinal);

        foreach (var control in spec.Dialogs.SelectMany(d => d.Controls).Where(IsInput))
        {
            var property = control.Property!;
            if (!seen.Add(property))
            {
                continue;
            }

            var value = Default(spec, property)
                ?? control.Items?.FirstOrDefault()?.Value
                ?? (control.Type == "CheckBox" ? "1" : null);
            parts.Add(value is null ? $"{property}=\"<value>\"" : $"{property}={Quote(value)}");
        }

        return string.Join(" ", parts);
    }

    private static bool IsInput(MockControl control) =>
        control.Property is not null && MockSpecValidator.InputTypes.Contains(control.Type);

    private static IEnumerable<string> Forms(MockControl control)
    {
        var property = control.Property!;
        if (control.Items is { Count: > 0 } items)
        {
            return items.Select(i => $"{property}={Quote(i.Value)}");
        }

        return control.Type switch
        {
            "CheckBox" => [$"{property}=1 (checked)", "omit for unchecked"],
            "Edit" when ((control.Attributes ?? 3) & ControlAttributes.Password) != 0 =>
                [$"{property}=\"…\"", "hidden in logs only when listed in MsiHiddenProperties"],
            _ => [$"{property}=\"…\""],
        };
    }

    private static string? Default(MockSpec spec, string property) => spec.Properties?.GetValueOrDefault(property);

    private static string Quote(string value) => value.Any(c => char.IsWhiteSpace(c) || c == '"') ? $"\"{value.Replace("\"", "\"\"")}\"" : value;

    private static string Encode(string text) => WebUtility.HtmlEncode(text);
}

using System.ComponentModel;
using System.Text.Json;
using System.Text.Json.Serialization;
using AdvancedInstallerMcp.Dialogs;
using AdvancedInstallerMcp.Lint;
using AdvancedInstallerMcp.Mock;
using AdvancedInstallerMcp.Msi;
using AdvancedInstallerMcp.Render;
using ModelContextProtocol;
using ModelContextProtocol.Protocol;
using ModelContextProtocol.Server;

namespace AdvancedInstallerMcp.Tools;

[McpServerToolType]
public static class MockDialogTool
{
    // Enums as names (Severity "Warning"), as lint_dialog's result shows them.
    private static readonly JsonSerializerOptions ReportJson = new(JsonSerializerDefaults.Web)
    {
        Converters = { new JsonStringEnumConverter() },
    };

    [McpServerTool(Name = "mock_dialog", ReadOnly = false, Destructive = false, Idempotent = true, OpenWorld = false)]
    [Description(
        "Shows proposed setup pages before any project is changed. Takes the pages as Windows Installer table rows " +
        "(Dialog, Control, RadioButton/ComboBox/ListBox items, ControlCondition, Property defaults), adds Advanced " +
        "Installer's wizard frame (banner, lines, Back/Next/Cancel) unless frame is 'none', writes a throw-away MSI and " +
        "draws every page the way render_dialog draws a built one, so only what Windows Installer can show appears. " +
        "Returns a picture per page and lint_dialog's findings (text that does not fit, overlaps, tab order, choices " +
        "without a default, a control that does not suit the number of choices). With outputDir it also writes the " +
        "pictures and index.html, an approval page with each page's inputs, why each control was chosen, and the " +
        "msiexec /qn command line: show it to the user and change the .aip only after they approve. Refuses control " +
        "types Windows Installer does not have and lower-case input properties (they cannot be set unattended).")]
    public static CallToolResult MockDialog(
        [Description("The proposed pages.")] MockSpec spec,
        [Description("Draw only this page. Omit for all.")] string? dialog = null,
        [Description("Display scaling in percent: 100, 125 or 150.")] int scaling = 100,
        [Description("Property values for another state of the pages, as in render_dialog. null or \"\" unsets.")]
        Dictionary<string, string?>? properties = null,
        [Description("Folder for <Dialog>.png and the approval page index.html. Omit to only return the pictures.")]
        string? outputDir = null)
    {
        var dpi = DisplayScaling.ToDpi(scaling);

        var problems = MockSpecValidator.Problems(spec);
        if (problems.Count > 0)
        {
            throw new McpException("The spec has problems:" + Environment.NewLine + string.Join(Environment.NewLine, problems.Select(p => "- " + p)));
        }

        var names = spec.Dialogs.Select(d => d.Name).ToList();
        if (dialog is not null && !names.Contains(dialog, StringComparer.Ordinal))
        {
            throw new McpException($"The spec has no page '{dialog}'. Pages: {string.Join(", ", names)}.");
        }

        var path = Path.Combine(Path.GetTempPath(), $"advinst-mock-{Guid.NewGuid():N}.msi");
        try
        {
            PackageWriter.Write(path, MockPackageBuilder.Build(spec));
            using var database = MsiDatabase.OpenReadOnly(path);
            var package = DialogPackage.Load(database);

            var shown = (dialog is null ? names : [dialog]).Select(n => package.FindDialog(n)!).ToList();
            var findings = DialogLinter.Lint(package, shown);
            var rendered = shown.Select(d => DialogRenderer.Render(package, d, dpi, properties ?? [])).ToList();

            string? approvalPage = null;
            if (outputDir is not null)
            {
                var dir = Path.GetFullPath(outputDir);
                Directory.CreateDirectory(dir);
                for (var i = 0; i < shown.Count; i++)
                {
                    File.WriteAllBytes(Path.Combine(dir, shown[i].Name + ".png"), rendered[i].Png);
                }

                approvalPage = Path.Combine(dir, "index.html");
                File.WriteAllText(approvalPage, ApprovalPage.Html(spec with { Dialogs = spec.Dialogs.Where(d => shown.Any(s => s.Name == d.Name)).ToList() }, findings));
            }

            var report = JsonSerializer.Serialize(new
            {
                dialogs = rendered.Select(r => r.Report),
                findings,
                approvalPage,
            }, ReportJson);

            return new CallToolResult
            {
                Content =
                [
                    .. rendered.Select(r => (ContentBlock)ImageContentBlock.FromBytes(r.Png, "image/png")),
                    new TextContentBlock { Text = report },
                ],
            };
        }
        finally
        {
            File.Delete(path);
        }
    }
}
